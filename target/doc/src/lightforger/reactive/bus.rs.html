1//! CompilerBus — multicast event bus for reactive compilation.
2//!
3//! Provides three multicast channels:
4//! - `var_events`: Value bind/update notifications
5//! - `module_events`: Module load and member registration
6//! - `css_scope_subject`: CSS scope open/close boundaries
7
8use rxrust::prelude::*;
9use std::convert::Infallible;
10use std::fmt;
11
12/// Events emitted when a Sass variable is bound or updated.
13#[derive(Debug, Clone)]
14pub enum ValueEvent {
15    /// A new variable binding: scope_id, name, value
16    Bind {
17        scope_id: u64,
18        name: String,
19        value: u64,
20    },
21}
22
23/// Events emitted when modules are loaded or members registered.
24#[derive(Debug, Clone)]
25pub enum ModuleEvent {
26    /// A module has been loaded by name.
27    Load { name: String },
28}
29
30/// Events emitted to mark CSS scope boundaries.
31#[derive(Debug, Clone)]
32pub enum ScopeEvent {
33    /// Enter a new scope.
34    Open { scope_id: u64 },
35    /// Exit the current scope.
36    Close { scope_id: u64 },
37}
38
39/// Multicast event bus shared across all pipeline stages.
40///
41/// Uses `Local` (single-threaded) subjects internally. The `CompilerBus`
42/// itself is `Clone` via subject cloning — clones share the same
43/// multicast channels.
44#[derive(Clone)]
45pub struct CompilerBus {
46    var_events: LocalSubject<'static, ValueEvent, Infallible>,
47    module_events: LocalSubject<'static, ModuleEvent, Infallible>,
48    css_scope_subject: LocalSubject<'static, ScopeEvent, Infallible>,
49}
50
51impl CompilerBus {
52    /// Create a new `CompilerBus` with fresh multicast subjects.
53    pub fn new() -> Self {
54        Self {
55            var_events: Local::subject(),
56            module_events: Local::subject(),
57            css_scope_subject: Local::subject(),
58        }
59    }
60
61    /// Subscribe to variable bind events.
62    pub fn var_events(&self) -> LocalSubject<'static, ValueEvent, Infallible> {
63        self.var_events.clone()
64    }
65
66    /// Subscribe to module events.
67    pub fn module_events(&self) -> LocalSubject<'static, ModuleEvent, Infallible> {
68        self.module_events.clone()
69    }
70
71    /// Subscribe to CSS scope events.
72    pub fn css_scope(&self) -> LocalSubject<'static, ScopeEvent, Infallible> {
73        self.css_scope_subject.clone()
74    }
75}
76
77impl fmt::Debug for CompilerBus {
78    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
79        f.debug_struct("CompilerBus")
80            .finish_non_exhaustive()
81    }
82}
83
84impl Default for CompilerBus {
85    fn default() -> Self {
86        Self::new()
87    }
