13/// CSS output statement produced by the evaluator.
14#[derive(Debug, Clone)]
15pub enum CssStmt {
16    /// A simple `property: value;` declaration.
17    Decl { property: String, value: String },
18    /// A wrapped media query with inner statements.
19    Media { query: String, inner: Vec<CssStmt> },
20    /// A wrapped supports query with inner statements.
21    Supports { query: String, inner: Vec<CssStmt> },
22    /// Inner style rule.
23    Rule { selector: String, inner: Vec<CssStmt> },
24}
25
26impl CssStmt {
27    /// Returns true if this statement produces no visible CSS output.
28    pub fn is_invisible(&self) -> bool {
29        match self {
30            CssStmt::Decl { .. } => false,
31            CssStmt::Media { inner, .. }
32            | CssStmt::Supports { inner, .. }
33            | CssStmt::Rule { inner, .. } => inner.iter().all(CssStmt::is_invisible),
34        }
35    }
36}

66/// Compiler evaluation context.
67///
68/// Replaces `Rc<RefCell<Environment>>` — holds only an immutable bus reference
69/// and the current scope id.
70#[derive(Clone)]
71pub struct EvalContext {
72    pub bus: CompilerBus,
73    pub scope_id: ScopeId,
74}
75
76impl EvalContext {
77    /// Create a new `EvalContext` for the given bus and scope.
78    pub fn new(bus: CompilerBus, scope_id: ScopeId) -> Self {
79        Self { bus, scope_id }
80    }
81
82    /// Create a child scope with a new scope id.
83    pub fn child_scope(&self, local_idx: u64) -> Self {
84        let child_id = self.scope_id * 1000 + local_idx;
85        Self {
86            bus: self.bus.clone(),
87            scope_id: child_id,
88        }
89    }
90
91    /// Subscribe to variable events relevant to the current scope.
92    ///
93    /// Returns a boxed local observable of `ValueEvent` items that are
94    /// in the current scope or any ancestor scope.
95    pub fn ver(
96        &self,
97        _name: &str,
98    ) -> LocalBoxedObservableClone<'static, ValueEvent, Infallible> {
99        let current = self.scope_id;
100        self.bus
101            .var_events()
102            .filter(move |evt| matches!(evt, ValueEvent::Bind { scope_id, .. } if *scope_id <= current))
103            .box_it_clone()
104    }
105}
106
107impl fmt::Debug for EvalContext {
108    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
109        f.debug_struct("EvalContext")
110            .field("scope_id", &self.scope_id)
111            .finish_non_exhaustive()
112    }
113}

115/// Trait implemented by each AST node that acts as a directive operator.
116///
117/// The operator signature is `AstNode → AstNode` (intermediate), not directly
118/// `AstNode → CssStmt`, because directives like `@for` may need multiple
119/// expansion passes.
120pub trait SassOp {
121    /// Consume `self` and produce a boxed operator function.
122    fn into_operator(
123        self,
124        ctx: Rc<EvalContext>,
125    ) -> Box<dyn Fn(AstStream) -> AstStream>;
126}

128/// An AST node in the reactive pipeline.
129#[derive(Debug, Clone)]
130pub enum AstNode {
131    /// A variable declaration: `$name: value;`
132    VariableDecl { name: String, value: Value },
133    /// A CSS style declaration: `property: value;`
134    StyleDecl { property: String, value: String },
135    /// `@media` rule with query and inner block.
136    Media { query: String, inner: Vec<AstNode> },
137    /// `@supports` rule with query and inner block.
138    Supports { query: String, inner: Vec<AstNode> },
139    /// A rule set: `{ ... }`
140    RuleSet {
141        selector: String,
142        inner: Vec<AstNode>,
143    },
144    /// `@if` conditional
145    If {
146        cond: Box<AstNode>,
147        then_branch: Vec<AstNode>,
148        else_branch: Vec<AstNode>,
149    },
150    /// `@for` loop: `@for $i from 1 through 3 { ... }`
151    For {
152        var: String,
153        from: f64,
154        through: f64,
155        body: Vec<AstNode>,
156    },
157    /// `@each` loop: `@each $item in $list { ... }`
158    Each {
159        var: String,
160        body: Vec<AstNode>,
161    },
162    /// `@while` loop: `@while $i < 10 { ... }`
163    While {
164        cond: Box<AstNode>,
165        body: Vec<AstNode>,
166    },
167    /// `@warn` directive — emits a diagnostic message.
168    Warn { message: String },
169    /// `@debug` directive — emits a debug-level expression evaluation.
170    Debug { expr: Box<AstNode>, message: String },
171    /// Lowered CSS statement — produced by operators as their final output.
172    Css(CssStmt),
173    /// Placeholder for unimplemented nodes.
174    Placeholder,
175}
